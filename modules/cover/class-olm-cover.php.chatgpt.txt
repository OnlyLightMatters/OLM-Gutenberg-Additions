<?php

namespace OLM\GutenbergAdditions;

if ( ! defined( 'ABSPATH' ) ) {
	exit;
}

class OLM_GA_Cover {

	/**
	 * Initialize the Cover module.
	 *
	 * @return void
	 */
	public static function init() {

		add_action( 'init', [ __CLASS__, 'register_block_extension' ] );
		add_action( 'enqueue_block_editor_assets', [ __CLASS__, 'enqueue_editor_assets' ] );
		add_action( 'wp_enqueue_scripts', [ __CLASS__, 'enqueue_frontend_assets' ] );

		add_filter( 'render_block_core/cover', [ __CLASS__, 'render_cover' ], 10, 2 );

	}

	/**
	 * Register additional Cover block attributes.
	 *
	 * @return void
	 */
	public static function register_block_extension() {

		register_block_type(
			'core/cover',
			[
				'attributes' => [
					'olmHref' => [
						'type'    => 'string',
						'default' => '',
					],
					'olmOpensInNewTab' => [
						'type'    => 'boolean',
						'default' => false,
					],
				],
			]
		);

	}

	/**
	 * Enqueue editor assets.
	 *
	 * @return void
	 */
	public static function enqueue_editor_assets() {

		wp_enqueue_script(
			'olm-ga-cover',
			OLM_GA_URL . 'modules/cover/olm-cover.js',
			[
				'wp-block-editor',
				'wp-blocks',
				'wp-components',
				'wp-compose',
				'wp-element',
				'wp-hooks',
				'wp-i18n',
			],
			OLM_GA_VERSION,
			true
		);

		wp_set_script_translations(
			'olm-ga-cover',
			'olm-gutenberg-additions',
			OLM_GA_PATH . 'languages'
		);

	}

	/**
	 * Enqueue frontend assets.
	 *
	 * @return void
	 */
	public static function enqueue_frontend_assets() {

		wp_enqueue_style(
			'olm-ga-cover',
			OLM_GA_URL . 'modules/cover/olm-cover.css',
			[],
			OLM_GA_VERSION
		);

	}

	/**
	 * Render linked Cover blocks.
	 *
	 * @param string $block_content Rendered HTML.
	 * @param array  $block         Parsed block.
	 *
	 * @return string
	 */
	public static function render_cover( $block_content, $block ) {

		$attributes = $block['attrs'] ?? [];

		if ( empty( $attributes['olmHref'] ) ) {
			return $block_content;
		}

		$href = esc_url( $attributes['olmHref'] );

		$target = ! empty( $attributes['olmOpensInNewTab'] )
			? ' target="_blank" rel="noopener noreferrer"'
			: '';

		$overlay = sprintf(
			'<a class="olm-cover-link-overlay" href="%1$s"%2$s aria-label="%3$s"></a>',
			$href,
			$target,
			esc_attr__( 'Cover link', 'olm-gutenberg-additions' )
		);

		return preg_replace(
			'/(<div\b[^>]*class="[^"]*\bwp-block-cover\b[^"]*"[^>]*>)/',
			'$1' . $overlay,
			$block_content,
			1
		);

	}

}
